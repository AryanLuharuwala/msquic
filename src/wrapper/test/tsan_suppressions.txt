# ThreadSanitizer suppressions for the msquic wrapper stress test.
#
# libmsquic.so is built WITHOUT instrumentation, so TSan cannot see the
# happens-before edges established by its internal locks/atomics and will
# emit false positives whose stacks live entirely inside libmsquic frames.
# We suppress reports attributed to those library-internal frames ONLY.
#
# We deliberately do NOT suppress anything in the wrapper's own translation
# unit (quic.cpp: StreamCallback / ConnectionCallback / quic_send / quic_close
# / quic_init / quic_cleanup). Those are the races this audit must surface and
# fix; a clean wrapper run is the pass criterion.

# Races whose reported stacks are entirely inside the uninstrumented library.
called_from_lib:libmsquic.so

# Some msquic worker internals (lock-free queues, datapath) appear by symbol
# even when partially inlined into the .so; match the library object.
race:libmsquic.so
